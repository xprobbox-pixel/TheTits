cmake_minimum_required(VERSION 3.24)
project(TheTits VERSION 0.3.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OSX_DEPLOYMENT_TARGET 11.0 CACHE STRING "Minimum macOS")
set(JUCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.deps/JUCE-8.0.6" CACHE PATH "JUCE 8.0.6 source")
if(NOT EXISTS "${JUCE_DIR}/CMakeLists.txt")
 include(FetchContent)
 FetchContent_Declare(JUCE URL https://codeload.github.com/juce-framework/JUCE/tar.gz/refs/tags/8.0.6 URL_HASH SHA256=95752f8e8b37d802b4c10182bc757de7d88f0c8899e879afa8798589be306a11)
 FetchContent_MakeAvailable(JUCE)
 FetchContent_GetProperties(JUCE SOURCE_DIR JUCE_FETCHED_SOURCE)
 set(JUCE_DIR "${JUCE_FETCHED_SOURCE}")
else()
 add_subdirectory("${JUCE_DIR}" juce)
endif()
set(FORMATS VST3 Standalone)
if(APPLE)
 list(APPEND FORMATS AU)
endif()
juce_add_plugin(TheTits COMPANY_NAME "House Rocks" BUNDLE_ID "audio.houserocks.thetits" PLUGIN_MANUFACTURER_CODE HsRk PLUGIN_CODE TtBs FORMATS ${FORMATS} PRODUCT_NAME "The Tits" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx Modulation AU_MAIN_TYPE kAudioUnitType_Effect)
juce_add_binary_data(TheTitsAssets SOURCES Resources/poster.png Resources/factory.json Resources/Roboto.ttf Resources/Roboto-Bold.ttf)
target_sources(TheTits PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp Source/DSP/Engine.cpp Source/Presets/PresetManager.cpp Source/UI/ArtworkRenderer.cpp)
target_include_directories(TheTits PUBLIC Source)
target_compile_definitions(TheTits PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(TheTits PRIVATE TheTitsAssets juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
add_executable(TheTitsTests)
set_target_properties(TheTitsTests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/TheTitsTests_artefacts/$<CONFIG>")
target_include_directories(TheTitsTests PRIVATE "${JUCE_DIR}/modules")
target_compile_definitions(TheTitsTests PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
target_sources(TheTitsTests PRIVATE Tests/Tests.cpp)
target_link_libraries(TheTitsTests PRIVATE TheTits TheTitsAssets)
enable_testing()
add_test(NAME DSPAndState COMMAND TheTitsTests)

if(APPLE)
 foreach(format AU Standalone)
  add_custom_command(TARGET TheTits_${format} POST_BUILD COMMAND /usr/bin/codesign --force --deep --sign - "$<TARGET_BUNDLE_DIR:TheTits_${format}>" VERBATIM)
 endforeach()
endif()

target_sources(TheTits_Standalone PRIVATE Source/StandaloneMain.cpp)
target_compile_definitions(TheTits_Standalone PRIVATE JUCE_USE_CUSTOM_PLUGIN_STANDALONE_APP=1)
